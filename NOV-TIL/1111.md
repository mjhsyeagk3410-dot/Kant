```sh
    def __str__(self):
        state = self.mood() if self.alive else "떠남"
        return (f"{self.name} {state}\n"
                f" 배고픔: {self.bar(self.hunger)} {self.hunger:>3}\n"
                f" 행복도: {self.bar(self.happy)} {self.happy:>3}\n"
                f" 에너지: {self.bar(self.energy)} {self.energy:>3\n")

    def mood(self)
        if self.hunger >= 80 or self.happy <= 20: return "😭"
        if self.happy >= 70 and self.hunger <= 40: return "😊"
        return "😑"

    def bar(self, value):
        # 0~100 값을 10칸의 막대로 변환
        filled = value // 10
        return "🟩" * filled + "⬜" * (10 - filled)

    # 공통 메서드 
    def _clamp(self, value):
        #값의 범위를 (0~100) 유지
        if (value < 0): return 0
        if (value > 100): return 100
        return value

    def _change(self, hunger=0, happy=0, energy=0):
        # 상태값은 반드시 이 메서드를 사용한다 
        self.hunger = self._clamp(self.hunger + hunger)
        self.happy = self._clamp(self.happy + happy)
        self.energy = self._clamp(self.energy + energy)

    # 밥주기
    def feed(self):
        if self.hunger <= 10:
            print(f"{self.name}: 배불러요~")
            return False
        self._change(hunger=-30, happy=5)
        print(f"{self.name}에게 밥을 줬습니다. 냠냠!!")

    # 놀아주기
    def play(self):
        if self.energy < 20:
            print(f"{self.name}: 너무 피곤해요...")
            return False
        self._change(hunger=10, happy=25, energy=-20)
        print(f"{self.name}와 신나게 놀았습니다.")
        return True

    # 잠자기 
    def sleep(self):
        self._change(hunger=5, energy=40)
        print(f"{self.name}가 잠들었어요. zZ")
        return True

    # 시간결과 
    def pass_time(self):
        # 하루 지날때 마다 상태값 자연 감소
        self._change(hunger=10, happy=-10, energy=-5)
        # 즐거운 이벤트를 확률계산(20%)
        if (random.randint(1, 5) == 1):
            self._change(happy=15)
            print(f"{self.name} 숨겨진 장난감을 찾았어요.")

        if self.hunger >= 100 or self.happy <= 0:
            self.alive = False
```

```sh
# cart.py 소스파일
from product import Product

class Cart:
    def __init__(self):
    # 빈 장바구니 생성
    self.items = {} # {상품: 수량}

    # 장바구니에 상품담기
    def add(self, product, qty):
        # 품절 상품은 담을 수 없다 
        if product.is_sold_out():
            print("품절된 상품입니다.")
            return

        # 이미 장바구니에 담겨있는 수량
        in_cart_qty = self.items.get(product, 0)
        # 더 담을 수 있는 수량
        extra_qty = product.stock - in_cart_qty
        if qty > extra_qty: # 재고 수량 범위를 벗어난 경우
            print(f"재고가 부족합니다. (담을 수 있는 수량: {extra_qty}개)")
            return

        self.items[product] = self.items.get(product, 0) + qty
        print(f"{product.name} {qty}개를 담았습니다.")

    # 장바구니에서 상품 제거(뺴기)
    def remove (self, product, qty):
        #장바구니에 없는 상품은 뺄 수 없음 
        if product not in self.items:
            print("장바구니에 해당 상품이 없습니다.")
            return

        # 담은 수량 이상 빼면 상품을 통째로 삭제 
        if qty >= self.items[product]:
            del self.items[product] # 해당 상품을 딕셔너리에서 삭제 
            print(f"{product.name}를 장바구니에서 모두 뺐습니다.")
        else:
            self.items[product] -= qty
            print(f"{product.name} {qty}개를 뺐습니다.")

    # 징비구니 비었는지 확인
    def is_empty(self):
        return len(self.items) == 0

    # 상품 합계 계산
    def subtotal(self):
        return sum(p.price * q for p, q in self.items.items())

    # 배송비
    def shipping(self):
        if self.is_empty(): return 0
        return 0 if self.subtotal() >= 30_000 else 3_000

    # 결제 금액 계산
    def total(self):
        return self.subtotal() + self.shipping()

    # 장바구니 내용 출력 
    def show(self):
        line = "-" * 40
        print(line)

        if self.is_empty():
            print("장바구니가 비어있습니다.")

        for p, q in self.items.items():
            #제품명 x 수량 = 총합계
            print(f"{p.name:<10} x {p.price * q:>12,}원")

        print(line)
        print(f"{'상품 합계':<10} {self.subtotal():>15,}원")
        print(f"{'배송비':<10} {self.shippint{}:>15,}원")
        print(f"{'결제 금액':<10} {self.total():>15,}원")
        print(line)

    def checkout(self):
        if self.is_empty(): return

        self.show()
        # 재고 감소 처리 
        for p, q in self.items.items():
            p.stock -= q

        # 결제 성공여부 SKIP
        print("결제가 완료되었습니다. 감사합니다!")
```

```sh
class Product:
    next_id = 1 # 클래스 변수

    # 새 상품의 초기 정보 저장
    def __init__ (self, name, price, stock):
        self.id = Product.next_id
        Product.next_id += 1 # 다음 상품을 위해 Index값+1
        self.name = name
        self.price = price
        self.stock = stock

    def is_sold_out(self):
        return self.stock == 0

    def __str__(self):
        stock_text = "품절" if self.is_sold_out() else f"재고 {self.stock}개"
        return f"{self.id}: {self.name:<10} {self.price:>8,}{stock_text}"
```

```md
01 클래스와 상속을 활용해 객체지향 방식으로 프로그램을 구현할 수 없다. 공부 필요!
02 리스트, 튜플, 딕셔너리, 집합의 특징과 차이를 이해하고 목적에 맞는 자료구조를 선택하여 데이터를 저장, 조회, 가공할 수 없다. 공부 필요! 
03 조건문과 반복문을 조합하여 프로그램의 실행 흐름을 제어하고 데이터 검사, 검색, 집계, 필터링 기능을 구현할 수 없다. 공부 필요!
04 함수, 매개변수, 반환값 및 변수 스코프를 이해하고, 반복되는 로직을 함수, 모듈, 패키지로 분리하여 재사용 가능한 코드로 구성할수 없다. 공부 필요!
05 구문 오류와 런타임 예외의 차이를 이해하고 예외 처리 구문과 디버깅 도구를 활용하여 오류의 원인을 진단하고 수정 할 수 없다. 공부 필요! 
06 거대한 요구사항을 작은 조각으로 쪼개는 '문제 분해력'도 연구와 학습이 많이 필요함
``` 
