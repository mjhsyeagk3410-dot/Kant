### 프로젝트에서 코드륾 만들때 주석을 잘 달아야한다
```md
왜? 주석은 미래에 이 코드를 볼 나 자신을 위한것이고 나와 함께하는 팀원들을 위한것이다. 

근데 주의할 점도 있다. 
뻔한 일을 다시 한 번 풀어쓰는 주석은 의미없다. 
코드를 고쳤다면 주석도 고쳐야한다. 틀린 주석은 없는 주석보다 나쁘다. 
코드를 주석 처리한채 방치하지 말자 필요없으면 지우자 어차피 버전관리가 기억한다. 

좋은 주석이라는것은, '왜?'를 설명한다.
어떤 제약때문에, 어떤 우회책을 썼고, 이런 코드만 봐서는 알 수 없는 결정같은것들도 쓰면 좋다. 
```
### enumerate() 에 대해서 알아보자 
```sh
# 튜터님이 가위바위보 게임만들때 승패 판결목록 표시를 위해 쓴 enumerate()
def show_records(recodes):
    print("\n[판별 승패 목록]")
    # "1판 : 가위 vs 바위 => 패배
    # "2판 : 가위 vs 바위 => 패배
    # "3판 : 가위 vs 바위 => 패배
    for number, record in enumerate(recodes, start=1):
        print(f"{number}판: {record}")
```
#### 왜 썼는가? 
```md
반복 횟수(몇 판째인지)와 리스트 안의 문자열 데이터를 동시에 추적해야 할 때, 코드를 길게 쓰고 수동으로 변수(number += 1)를 조작할 필요 없이 가장 파이썬스럽고 깔끔하게 해결하기 위해 사용한다
```
### 디버깅은 왜 해여하는걸까?
```md
디버깅은 개발자의 실력을 결정짓는 핵심 역량이며 단순한 에러 수정이 아니라 코드를 이해하고 문제의 원인을 분석하는 모든 과정이다. 

디버깅은 단순한 기술이 아니라 개발자의 사고력과 문제해결 능력이 드러나는 영역이며 디버깅을 잘하면 코드 품질이 올라가고 실무에서도 신뢰받는 개발자가 되며 코드는 작성보다 수정이 더 많다 곧 디버깅은 개발의 반이다 

디버깅을 잘한다는 것은, 코드 흐름을 머리속에 그릴 수 있어야한다. 또한 에러 메시지를 무시하지 말고 의미를 해석 할 수 있어야 한다. vscode외에도 다양한 디버깅 도구를 익히고 활용해 볼 수 있어야 한다. 
```

```sh
# main.py

import random

class Pet:
    # 클래스 변수
    count = 0 # 지금까지 입양한 펫 수를 기록

    def __init__(self, name):
        self.name = name # 이름
        self.hunger = 30 # 배고픔(0~100)
        self.happy = 70 # 행복도(0~100)
        self.energy = 70 # 에너지(0~100)
        self.alive = True
        Pet.count += 1 

    def __str__(self):
        state = self.mood() if self.alive else "떠남"
        return (f"{self.name} {state}\n"
                f"배고픔: {self.bar(self.hunger)} {self.hunger:>3}\n"
                f"행복도: {self.bar(self.happy) {self.happy:>3}\n"
                f"에너지: {self.bar(self.energy)} {self.energy:>3}\n")
    def mood(self):
        if self.hunger >= 80 or self.happy <= 20: return "😭"
        if self.happy >= 70 and self.hunger <= 40: return "😊"
        return "😑"

    def bar(self, value):
        #0~100값을 10칸의 막대로 변환
        filled = value // 10
        return "*" * filled + "-" * (10 - filled)

    #공통 메서드
    def _clamp(self, value):
        # 값의 범위를 (0~100) 유지 
        if (value < 0): return 0
        if (value > 100): return 100
        return value

    def change(self, hunger=0, happy=0, energy=0):
        # 상태값은 반드시 이 메서드를 사용한다 
        self.hunger = self._clamp(self.hunger + hunger)
        self.happy = self._clamp(self.happy + happy)
        self.energy = self._clamp(self.energy + energy)

    # 밥주기
    def feed(self):
        if self.hunger <= 10:
            print(f"{self.name}: 배불러요~")
            return False
        self._change(hunger=-30, happy=5)
        print(f"{self.name}에게 밥을 줬습니다. 냠냠!!")

    # 놀아주기 
    def play(self):
        if self.energy < 20:
            print(f"{self.name}: 너무 피곤해요...")
            return False
        self._change(hunger=10, happy=25, energy=20)
        print(f"{self.name}와 신나게 놀았습니다.")
        return True

    # 잠자기
    def sleep(self):
        self._change(hunger=5, energy=40)
        print(f"{self.name}가 잠들었어요. zZ")
        return True

    # 시간 결과 
    def pass_time(self):
        # 하루 지날때 마다 상태값 자연 감소 
        self._change(hunger=10, happy=-10, energy=-5)
        # 즐거운 이벤트를 확률계산(20%)
        if(random.randint(1, 5) == 1):
            self._change(happy=15)
            print(f"{self.name} 숨겨진 장난감을 찾았어요.")

        if self.hunger >= 100 or self.happy <= 0:
            self.alive = False

# -------- 테스트 코드 ----------

    choco.pass_time()
    print(f"{day}일차 : 배고픔 {choco.hunger}, 행복도 {choco.happy}")
    day += 1
```

```sh
# cart.py
from product import Product

class Cart:
    def __init__(self):
    # 빈 장바구니 생성
    self.items = {} # {상품: 수량}

    # 장바구니에 상품담기 
    def add(self, product, qty):
        # 품질 상품은 담을 수 없다
        if product.is_sold_out():
            print("품절된 상품입니다.")
            return

        # 이미 장바구니에 담겨있는 수량 
        in_cart_qty = self.items.get(product, 0)
        # 더 담을 수 있는 수량 
        extra_qty = product.stock - in_cart_qty
        if qty > extra_qty: # 재고 수량범위를 벗어난 경우 
            print(f"재고가 부족합니다. (담을 수 있는 수량: {extra_qty}개)")
            return

        self.items[product] = self.items.get(product, 0) + qty
        print(f"{product.name} {qty}개를 담았습니다.")

    # 장바구니에서 상품 제거(빼기)
    def remove (self, product, qty):
        # 장바구니에 없는 상품은 뺄 수 없음
        if product not in self.items:
            print("장바구니에 해당 상품이 없습니다.")
            return

        # 담은 수량 이상 빼면 상품을 통째로 삭제
        if qty >= self.items[product]:
            del self.items[product] # 해당 상품을 딕셔너리에서 삭제 
            print(f"{product.name}를 장바구니에서 모두 뺐습니다.")
        else: 
            self.items[product] -= qty
            print(f"{product.name} {qty}개를 뺐습니다.")

    # 장바구니 비었는지 확인
    def is_empty(self):
        return len(self.items) == 0

    # 상품 합계 계산
    def subtotal(self):
        return sum(p.price * q for p, q in self.items.items())

    # 배송비
    def shipping(self):
        if self.is_empty(): return 0
        return 0 if self.subtotal() >= 30_000 else 3_000

    # 결제 금액 계산
    def total(self):
        return self.subtotal() + self.shipping()

    # 장바구니 내용 출력
    def show(self):
        line = "-" * 40
        print(line)

        if self.is_empty():
            print("장바구니가 비어있습니다.")

        for p, q in self.items.items():
            # 제품명 x 수량 = 총합계 
            print(f"{p.name:<10} x {p.price * q:>12,}원")

        print(line)
        print(f"{'상품 합계':<10} {self.subtotal():>15,}원")
        print(f"{'배송비':<10} {self.shipping{}:>15,}원")
        print(f"{'결제 금액':<10} {self.total():>15,}원")
        print(line)

    def checkout(self):
        if self.is_empty(): return

        self.show()
        # 재고 감소 처리 
        for p, q in self.items.items():
            p.stock -= q 

        # 결제 성공여부 SKIP
        print("결제가 완료되었습니다. 감사합니다!")
```

```sh
class Product:
    next_id = 1 # 클래스 변수

    # 새 상품의 초기 정보 저장 
    def __init__ (self, name, price, stock):
        self.id = Product.next_id
        Product.next_id += 1 # 다음 상품을 위해 Index값+1
        self.name = name
        self.price = price
        self.stock = stock

    def is_sold_out(self):
        return self.stock == 0

    def __str__(self):

        return f"{self.id}: {self.name:<10} {self.price:>8,}{stock_text}"
```
