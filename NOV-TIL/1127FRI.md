```sh
df = pd.DataFrame({
    "study_hours":    [1, 2, 3, 4, 5, 2, 3, 5, 4, 1],
    "sleep_hours":    [6, 8, 5, 7, 8, 5, 8, 6, 6, 8],
```

```sh
X = df[["study_hours", "sleep_hours", "practice_count"]]
# 전체 데이터(df)에서 예측의 근거가 될 공부시간, 수면시간, 연습횟수 세 가지 항목만 뽑아내 입력데이터에 저장

y = df["score"] >= 60
# 전체 데이터(df)에서 점수가 60점 이상인지 확인해 60점 이상이면 True(합격), 미만이면 False(불합격)로 변환한 뒤 정답 데이터(y)에 저장. 
```

```sh
print(f"학생 {len(X)} 명, X : {X.shape[1]}개")
# 입력 데이터(X)를 기준으로 전체 학생 수(len(X)와 예측에 쓸 항목(특성)의 개수(X.shape[1], 여기서는 3개)를 "학생 00명, X:3개" 형태로 출력

# 원래 점수(score)열만 따로 떼어낸 뒤 그 옆에 60점 이상 여부를 담은 passed 열(y 값, True/False)을 임시로 붙여서 앞에서부터 5줄(head())만 미리 보여줌. 원본 데이터(df) 자체는 바뀌지 않음 
```

```sh
# 데이터 분리 및 표준화 
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
```
<br>

|||
| --- | --- |
| from | 모듈이나 패키지에서 원하는 함수, 클래스, 변수만 골라 가져올때 | 
| sklearn | scikit-learn은 numpy, scipy기반으로 구축된 파이썬 대표 오픈소스 머신러닝 라이브러리 | 
| model_selection | 주어진 데이터와 문제에 가장 잘 맞는 알고리즘과 하이퍼파라미터를 찾고 일반화 성능을 검증하는 과정 | 
| train_test_split | 머신러닝 모델을 평가하기 위해 전체 데이터셋을 훈련데이터와 테스트 데이터로 무작위 분할할때 사용하는 sklearn의 핵심기능 |
|||
<br>

```sh
# 학습용 vs 테스트용 데이터 분할
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=42, stratify=y
)

# 표준화 
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

print(f"학습 {len(X_train)}명, 테스트 {len(X_test)}명")
# iloc[행, 열] 
print(f"첫 학습 학생의 공부시간: {X_train.iloc[0, 0]} 시간")
print(f"표준화 값 {X_train_scaled[0, 0]:.2f}")

# 모델학습

from sklearn.linear_model import LogisticRegression

# 표준화된 학습데이터 + 정답
model = LogisticRegression(max_iter=1000)
model.fit(X_train_scaled, y_train)

print("클래스 :", model.classes_.tolist())
print("가중치 :", model.coef_[0].round(3))

# 새로운 학생 데이터로 예측
new = pd.DataFrame([[4, 7, 30]], columns=X.columns)
new_scaled = scaler.transform(new)

predict = model.predict(new_scaled)[0]

pass_proba = model.predict_proba(new_scaled)[0, 1]

print("예측 결과:", "합격" if predict else "불합격")
print(f"합격 확률: {pass_proba:.2%}")

#
from sklearn.metrics import accuracy_score, confusion_matrix

# 학습에서 사용하지 않은 학생 3명으로 예측, 평가
test_pred = model.predict(X_test_scaled)

print("실제 정답: ", y_test.tolist())
print("예측 결과: ", test_pred.tolist())
print(f"정확도: {accuracy_score(y_test, test_pred):.4%}")
print(confusion_matrix(y_test, test_pred, labels=[False, True]))
```
