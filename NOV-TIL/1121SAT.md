import random

TARGET_WINS = 3

def play_rps():
    choices = ["가위", "바위", "보"]
    # 이기는 조합 매핑: {내 선택: 내가 이기는 상대 선택}
    win_map = {"가위": "보", "바위": "가위", "보": "바위"}

    user_score = 0
    com_score = 0
    round_num = 0

    print(
        f"=== 가위바위보 3선승제 시작! ({TARGET_WINS}승 먼저 달성 시 우승, '종료' 또는 'q'로 중단) ==="
    )

    while user_score < TARGET_WINS and com_score < TARGET_WINS:
        user_choice = input("\n가위, 바위, 보 중 하나를 입력하세요: ").strip()

        if user_choice in ("종료", "q", "Q", "exit"):
            print("\n게임을 종료합니다.")
            break

        if user_choice not in choices:
            print("잘못된 입력입니다. '가위', '바위', '보' 중에서 입력해주세요.")
            continue

        round_num += 1
        com_choice = random.choice(choices)

        print(f"[{round_num}라운드] 유저: {user_choice} vs 컴퓨터: {com_choice}")

        if user_choice == com_choice:
            print("결과: 무승부!")
        elif win_map[user_choice] == com_choice:
            user_score += 1
            print("결과: 유저 승리!")
        else:
            com_score += 1
            print("결과: 컴퓨터 승리!")

        print(
            f"현재 스코어 -> 유저 {user_score} vs {com_score} 컴퓨터 (목표: {TARGET_WINS}승)"
            )

    print("\n=== 최종 스코어 ===")
    print(f"최종 스코어: 유저 {user_score} vs {com_score} 컴퓨터 (총 {round_num}판 진행)")
    
    if user_score >= TARGET_WINS:
        print("🎉 축하합니다! 유저가 3승을 먼저 달성해 최종 승리했습니다!")
    elif com_score >= TARGET_WINS:
        print("💻 아쉽네요! 컴퓨터가 3승을 먼저 달성해 최종 승리했습니다!")
    else:
        print("경기가 중단되어 최종 승자가 결정되지 않았습니다.")

if __name__ == "__main__":
    play_rps()
